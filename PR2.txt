-- ----------------------------------------------------- 1 INNER JOIN -----------------------------------------------------
SELECT * FROM CUSTOMERS INNER JOIN ORDERS ON CUSTOMERS.CUSTOMER_ID = CUSTOMERS.CUSTOMER_ID;
-- ----------------------------------------------------- 2 LEFT JOIN -----------------------------------------------------
SELECT * FROM CUSTOMERS LEFT JOIN ORDERS ON CUSTOMERS.CUSTOMER_ID = CUSTOMERS.CUSTOMER_ID;
-- ----------------------------------------------------- 3 RIGHT JOIN -----------------------------------------------------
SELECT * FROM CUSTOMERS RIGHT JOIN ORDERS ON CUSTOMERS.CUSTOMER_ID = CUSTOMERS.CUSTOMER_ID;
-- ----------------------------------------------------- 4 FULL OUTER JOIN -----------------------------------------------------
SELECT * FROM CUSTOMERS LEFT JOIN ORDERS ON CUSTOMERS.CUSTOMER_ID = CUSTOMERS.CUSTOMER_ID
UNION
SELECT * FROM CUSTOMERS RIGHT JOIN ORDERS ON CUSTOMERS.CUSTOMER_ID = CUSTOMERS.CUSTOMER_ID;
-- ----------------------------------------------------- 5 Subquery  -----------------------------------------------------
SELECT DISTINCT c.CUSTOMER_ID, c.FIRST_NAME, c.LAST_NAME
FROM CUSTOMERS c
JOIN ORDERS o ON c.CUSTOMER_ID = o.CUSTOMER_ID
WHERE o.TOTAL_AMOUNT >
      (SELECT AVG(TOTAL_AMOUNT) FROM ORDERS);

-- ----------------------------------------------------- 6 Subquery -----------------------------------------------------
SELECT EMPLOYEE_ID, FIRST_NAME, LAST_NAME, SALARY
FROM EMPLOYEES
WHERE SALARY > (SELECT AVG(SALARY) FROM EMPLOYEES);

-- ----------------------------------------------------- 7 Extract Year and Month from OrderDate -----------------------------------------------------
SELECT ORDER_ID,
       YEAR(ORDER_DATE) AS Order_Year,
       MONTH(ORDER_DATE) AS Order_Month
FROM ORDERS;

-- ----------------------------------------------------- 8 Difference in Days -----------------------------------------------------
SELECT ORDER_ID,
       DATEDIFF(CURDATE(), ORDER_DATE) AS Days_Difference
FROM ORDERS;

-- ----------------------------------------------------- 9 Format OrderDate -----------------------------------------------------
SELECT ORDER_ID,
       DATE_FORMAT(ORDER_DATE, '%d-%b-%Y') AS Formatted_Date
FROM ORDERS;

-- ----------------------------------------------------- 10 Concatenate FirstName & LastName -----------------------------------------------------
SELECT CUSTOMER_ID,
       CONCAT(FIRST_NAME, ' ', LAST_NAME) AS Full_Name
FROM CUSTOMERS;

-- ----------------------------------------------------- 11 Replace part of a string -----------------------------------------------------
SELECT FIRST_NAME,
       REPLACE(FIRST_NAME, 'Shah', 'Jonathan') AS Replaced_Name
FROM CUSTOMERS;

-- ----------------------------------------------------- 12 Uppercase FirstName & Lowercase LastName -----------------------------------------------------
SELECT CUSTOMER_ID,
       UPPER(FIRST_NAME) AS FirstName_Upper,
       LOWER(LAST_NAME) AS LastName_Lower
FROM CUSTOMERS;

-- ----------------------------------------------------- 13 Trim extra spaces from Email -----------------------------------------------------
SELECT CUSTOMER_ID,
       TRIM(EMAIL) AS Clean_Email
FROM CUSTOMERS;

-- ----------------------------------------------------- 14 Running Total of TotalAmount -----------------------------------------------------
SELECT ORDER_ID,
       TOTAL_AMOUNT,
       SUM(TOTAL_AMOUNT) OVER (ORDER BY ORDER_ID) AS Running_Total
FROM ORDERS;
	
-- ----------------------------------------------------- 15 Rank Orders by TotalAmount -----------------------------------------------------
SELECT ORDER_ID,
       TOTAL_AMOUNT,
       RANK() OVER (ORDER BY TOTAL_AMOUNT DESC) AS Amount_Rank
FROM ORDERS;

-- ----------------------------------------------------- 16 Assign Discount Based on TotalAmount -----------------------------------------------------
SELECT ORDER_ID,
       TOTAL_AMOUNT,
       CASE
           WHEN TOTAL_AMOUNT > 10000 THEN '10% Discount'
           WHEN TOTAL_AMOUNT > 500 THEN '5% Discount'
           ELSE 'No Discount'
       END AS Discount
FROM ORDERS;

-- ----------------------------------------------------- 17 Categorize Employees’ Salaries -----------------------------------------------------
    
SELECT EMPLOYEE_ID,
       FIRST_NAME,
       LAST_NAME,
       SALARY,
       CASE
           WHEN SALARY >= 150000 THEN 'High'
           WHEN SALARY >= 130000 THEN 'Medium'
           ELSE 'Low'
       END AS Salary_Level
FROM EMPLOYEES;
